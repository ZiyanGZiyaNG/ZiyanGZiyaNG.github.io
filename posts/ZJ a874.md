#include <bits/stdc++.h>
using namespace std;
int n;
typedef long long int ll;
typedef pair<ll, ll> pii;
const ll INF = 1e18;
vector<vector<pii>> g;
vector<ll> dijkstra(int s)
{
	vector<ll> d(26, INF);
	d[s] = 0;
	priority_queue<pii, vector<pii> , greater<pii>> pq;
	pq.push({d[s], s});
	while (!pq.empty())
	{
		ll dist = pq.top().first;
		int u = pq.top().second;
		pq.pop();
		if (dist > d[u]) continue;
		for (auto &edge : g[u])
		{
			int v = edge.first;
			ll w = edge.second;
			if (d[u] + w < d[v])
			{
				d[v] = d[u] + w;
				pq.push({d[v], v});
			}
		}
	}
	return d;
}

int main()
{
	ios::sync_with_stdio(false);
	cin.tie(0);
	
	while (cin >> n)
	{
		g.clear();
		g.resize(26);
		for (int i = 0; i < n; i++)
		{
			char a, b; cin >> a >> b;
			int c; cin >> c;
			int u = a - 'A';
			int v = b - 'A';
			g[u].push_back({v, c});
			g[v].push_back({u, c});
		}
		char s, e; cin >> s >> e;
		vector<ll> distances = dijkstra(s - 'A');
		if (distances[e - 'A'] == INF) cout << "NoRoute\n";
		else cout << distances[e - 'A'] << '\n';
	}	
}
