#include <bits/stdc++.h>
using namespace std;
typedef long long int ll;
typedef pair<ll, ll> pii;
const ll INF = 1e18;
vector<vector<pii>> g;
int n, m, s, t; 
vector<ll> dijkstra(int s)
{
	vector<ll> d(n, INF);
	d[s] = 0;
	priority_queue<pii, vector<pii>, greater<pii>> pq;
	pq.push({0, s});
	while (!pq.empty())
	{
		ll dist = pq.top().first;
		int u = pq.top().second; pq.pop();
		if (dist > d[u]) continue;
		for (auto &edge : g[u])
		{
			int v = edge.first;
			ll w = edge.second;
			if (d[u] + w < d[v])
			{
				d[v] = d[u] + w;
				pq.push({d[v], v});
			}
		}
	}
	return d;
}
int main()
{
	ios::sync_with_stdio(false);
	cin.tie(0);
	int y; cin >> y;
	for (int x = 1; x <= y; x++)
	{
		cin >> n >> m >> s >> t;
		g.clear();
		g.resize(n);
		for (int i = 0; i < m; i++)
		{
			int u, v, w; cin >> u >> v >> w;
			g[u].push_back({v, w});
			g[v].push_back({u, w});
		}
		cout << "Case #" << x << ": ";
		vector<ll> distances = dijkstra(s);
		if (distances[t] == INF) cout << "unreachable\n";
		else cout << distances[t] << '\n';
	}
}
